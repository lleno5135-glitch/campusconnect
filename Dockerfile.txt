FROM mcr.microsoft.com/dotnet/sdk:10.0 AS build
WORKDIR /src

COPY ["CampusConnect.csproj", "./"]
RUN dotnet restore "CampusConnect.csproj"

COPY . .
RUN dotnet publish "CampusConnect.csproj" -c Release -o /app/publish --no-restore

FROM mcr.microsoft.com/dotnet/aspnet:10.0 AS final
WORKDIR /app

COPY --from=build /app/publish .
EXPOSE 10000

CMD ["sh", "-c", "dotnet CampusConnect.dll --urls http://0.0.0.0:${PORT:-10000}"]
